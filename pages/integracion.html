Ejemplos de integración

// EcoSync SDK — Integración JavaScript
// NOTA: Simulación local — no hay servidor real en esta demo

const ecosync = {
  token: 'ecosync_pub_demo_tk_2024_abc123',
  baseUrl: 'https://api.ecosync.cl/v1',  // demo — no opera

  async syncDraft(draftData) {
    // En producción, esto haría un POST real al servidor
    const response = await fetch(`${this.baseUrl}/drafts/sync`, {
      method: 'POST',
      headers: {
        'Authorization': `Bearer ${this.token}`,
        'Content-Type': 'application/json'
      },
      body: JSON.stringify({
        draft: draftData,
        clientId: 'demo-client-001',
        timestamp: new Date().toISOString()
      })
    });
    return response.json();
  }
};

// Uso:
const draft = window.EcoSync.storage.loadDraft();
await ecosync.syncDraft(draft);
